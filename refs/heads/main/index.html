CLOUDFLARE WORKER DEPLOYMENT SCRIPT

Copy this JavaScript into your Cloudflare Worker to serve GitHub files as rendered HTML

export default {
  async fetch(request, env, ctx) {
    // 1. Raw GitHub repository URL of your HTML stream layout
    const githubRawUrl = "https://raw.githubusercontent.com/johannes837/Stream-layout/refs/heads/main/index.html";
    
    try {
      // 2. Fetch raw HTML from GitHub
      const response = await fetch(githubRawUrl);
      
      if (!response.ok) {
        return new Response("Failed to fetch stream layout from GitHub.", { status: 500 });
      }

      const htmlContent = await response.text();

      // 3. Return response with proper text/html Content-Type header so OBS renders it
      return new Response(htmlContent, {
        status: 200,
        headers: {
          "Content-Type": "text/html;charset=UTF-8",
          "Cache-Control": "public, max-age=30",
          "Access-Control-Allow-Origin": "*",
          "X-Frame-Options": "ALLOWALL"
        },
      });
    } catch (err) {
      return new Response(`Worker Error: ${err.message}`, { status: 500 });
    }
  },
};
